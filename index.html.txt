  function selectMovie(selectedOffset) {
    const winner = currentRoundMovies[matchIndex * 2 + selectedOffset];
    nextRoundMovies.push(winner);

    matchIndex++;

    if (matchIndex * 2 >= currentRoundMovies.length) {
      if (nextRoundMovies.length === 1) {
        showWinner(nextRoundMovies[0]);
      } else {
        currentRoundMovies = [...nextRoundMovies];
        nextRoundMovies = [];
        matchIndex = 0;
        renderMatch();
      }
    } else {
      renderMatch();
    }
  }
